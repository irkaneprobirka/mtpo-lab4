\begin{frame}{Постановка проблемы}
  \begin{block}{Проблема}
    В JavaScript-проектах выбор фреймворка модульного тестирования влияет не только на синтаксис тестов, но и на время запуска, потребление ОЗУ и пиковую загрузку CPU.
  \end{block}

  \textbf{Цель:} сравнить Jest, Vitest и Mocha на одинаковых тестах алгоритма Манакера.

  \vspace{0.25cm}
  Для каждого фреймворка измеряются:
  \[
    \tau_f(D) \text{ --- время}, \quad
    m_f(D) \text{ --- память}, \quad
    c_f(D) \text{ --- CPU}.
  \]

  Итоговый показатель:
  \[
    Score(f)=\frac{1}{3}\left(
    \frac{\overline{\tau}_f}{\max \overline{\tau}}+
    \frac{\overline{m}_f}{\max \overline{m}}+
    \frac{\overline{c}_f}{\max \overline{c}}
    \right).
  \]
\end{frame}

\begin{frame}{Тестируемый алгоритм: Манакер}
  \begin{columns}[T]
    \begin{column}{0.52\textwidth}
      \begin{itemize}
        \item Находит палиндромные подстроки за $O(n)$.
        \item Использует преобразованную строку с разделителями.
        \item Поддерживает текущий центр и правую границу.
        \item Подходит для масштабируемых входных строк.
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{block}{Преобразование}
        \[
          abba \Rightarrow \texttt{\^{}\#a\#b\#b\#a\#\$}
        \]
      \end{block}
      \begin{block}{Результат}
        Самый длинный палиндром: \texttt{abba}.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Технология тестирования: псевдокод метода}
  \small
  \begin{block}{Метод сравнения}
    \begin{enumerate}
      \item Загрузить тестовые строки из JSON или CSV.
      \item Проверить эталонный результат алгоритмом Манакера.
      \item Для каждого $f \in \{\text{Jest},\text{Vitest},\text{Mocha}\}$ выполнить $R=5$ запусков.
      \item Запустить тесты фреймворка как дочерний Node.js-процесс.
      \item Пока процесс выполняется, сэмплировать RSS-память и CPU по PID.
      \item Зафиксировать время, пиковую память, пиковый CPU и код завершения.
      \item Сохранить результаты в CSV и JSON.
    \end{enumerate}
  \end{block}
\end{frame}

\begin{frame}{Иллюстративный пример}
  \textbf{Эталонная строка:} \texttt{abacaba}

  \vspace{0.2cm}
  \begin{itemize}
    \item Преобразованная строка:
      \[
        \texttt{\^{}\#a\#b\#a\#c\#a\#b\#a\#\$}
      \]
    \item Центральный символ \texttt{c} становится центром максимального палиндрома.
    \item Самый длинный палиндром: \texttt{abacaba}.
    \item Количество палиндромных подстрок: 12.
  \end{itemize}
\end{frame}

\begin{frame}{Пошаговое выполнение на данных \texttt{abba}}
  \begin{enumerate}
    \item Вход: \texttt{abba}.
    \item Преобразование: \texttt{\^{}\#a\#b\#b\#a\#\$}.
    \item Алгоритм идет слева направо и считает радиусы.
    \item На разделителе между \texttt{b} и \texttt{b} радиус расширяется до всей строки.
    \item Обновляются $center$ и $right$.
    \item Результат: \texttt{longest = abba}, \texttt{count = 6}.
  \end{enumerate}
\end{frame}

\begin{frame}{Выводы}
  \begin{enumerate}
    \item Все обязательные проверки алгоритма Манакера прошли успешно.
    \item В данной алгоритмической задаче Mocha оказался самым быстрым и легким.
    \item Jest занял промежуточную позицию, но дает богатую встроенную экосистему.
    \item Vitest показал высокие накладные расходы в этой конфигурации, но полезен для Vite/ESM-проектов.
    \item Главный результат --- воспроизводимая методика сравнения фреймворков на конкретном проекте.
  \end{enumerate}
\end{frame}
